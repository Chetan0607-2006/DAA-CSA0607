FUNCTION SORT_ALERTS(alerts)
    RETURN stable sort of alerts by severity in descending order
    // Stability preserves original arrival order for equal severities.
END FUNCTION

Reference insertion-sort logic:
FOR i = 1 TO length(alerts)-1
    key = alerts[i]; j = i-1
    WHILE j >= 0 AND alerts[j].severity < key.severity
        alerts[j+1] = alerts[j]
        j = j-1
    END WHILE
    alerts[j+1] = key
END FOR
